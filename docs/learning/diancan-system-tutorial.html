为什么要学这个？ 你不读懂整体就改代码，就像不开地图就上路——容易迷路。U1 给你一张"地图"，告诉你三端各自在哪、它们怎么说话。
🍽️ 生活类比：一家餐厅的三个部门 把这套系统想象成一家餐厅：
  • 后端 = 厨房 + 中央系统：决定能不能做菜、定价多少、扣多少积分
  • 管理端 = 前台 + 后勤办公室：服务员点单、收银员结账、店长看报表
  • 小程序 = 顾客手里的菜单：扫码点单、查会员、领券
三部门通过"单据"沟通：顾客递给服务员一个订单，服务员递给厨房一个出餐单，厨房回执递回给服务员。这套"单据流转"就是后端 API。

  • 三端单体仓库：一个 Git 仓库里三个独立子项目（diancan-admin / diancan-admin-web / diancan-miniapp）。
  • 统一上下文路径 /api：后端所有接口都以 /api 开头，例如 /api/admin/dish/list。
  • 双端前缀 /admin 与 /app：管理端走 /api/admin/*，小程序走 /api/app/*，支付回调走 /api/notify/*。
  • 中间件四件套：MySQL（事实库）、Redis（会话+缓存）、RocketMQ（可靠消息）、MinIO（文件存储）。

// 文件：diancan-admin-web/src/service/request/index.ts
import axios, { type AxiosInstance, type AxiosRequestConfig } from 'axios'
import { getToken } from '@/utils/auth'

const service: AxiosInstance = axios.create({
  baseURL: import.meta.env.VITE_SERVICE_BASE_URL,  // 自动读 .env.dev
  timeout: 15000,
})

// 请求拦截器：自动塞 Token
service.interceptors.request.use((config) => {
  const token = getToken()
  if (token) {
    config.headers.Authorization = `Bearer ${token}`
  }
  return config
})

// 响应拦截器：统一处理错误码
service.interceptors.response.use(
  (response) => {
    const { code, msg, data } = response.data
    if (code === 0) return data          // 成功：返回数据
    if (code === 401) {
      // Sa-Token 鉴权失败：跳转登录
      window.location.href = '/login'
    }
    // 业务错误：抛错给上层处理
    return Promise.reject(new Error(msg || '请求失败'))
  },
  (error) => Promise.reject(error)
)

export default service

为什么要学这个？ 对后端工程师来说，小程序代码相对陌生。但读懂小程序只需要把握"一个请求入口 + 一个页面生命周期"两件事。
📖 类比：小程序就像一本翻页书 每一页 = 一个页面（pages/xxx）；书脊 = app.js（全局状态）；目录 = app.json（页面注册）。utils/request.js 是"由你联系作者"的统一通道。

  • app.json：小程序的"目录"，注册所有页面和 tabBar；小程序启动时第一读它。
  • utils/request.js：所有 HTTP 请求的单一入口，统一加前缀、Token、错误处理。
  • config/env.js：环境变量（API 地址），真机调试要改这里。
  • pages/ 一个页面 = 4 个文件：.js .json .wxml .wxss（类比 HTML 项目的 html/css/js 分离）。

const { API_HOST, apiPrefix } = require('../config/env.js')

function request(url, method, data, options = {}) {
  const token = wx.getStorageSync('token')
  return new Promise((resolve, reject) => {
    wx.request({
      url: API_HOST + apiPrefix + url,   // 拼成完整 URL
      method,

@Service
@RequiredArgsConstructor
public class OrderServiceImpl implements OrderService {

    private final OrderMapper orderMapper;
    private final CartMapper cartMapper;
    private final DishMapper dishMapper;
    private final TableMapper tableMapper;
    private final CouponService couponService;
    private final MemberSettlementService memberSettlementService;

    @Override
    @Transactional(rollbackFor = Exception.class)
    public OrderVO create(OrderCreateDTO dto) {
        // 1. 校验桌台
        Table table = tableMapper.selectById(dto.getTableId());
        if (table == null || table.getStatus() != 1) {
            throw new BizException("桌台不可用");
        }

        // 2. 加载购物车
        List<CartItem> items = cartMapper.selectByUserId(dto.getUserId());
        if (items.isEmpty()) {
            throw new BizException("购物车为空");
        }

        // 3. 锁库存（重要！防止超卖）
        for (CartItem item : items) {
            int updated = dishMapper.lockStock(item.getDishId(), item.getQuantity());
            if (updated == 0) {
                throw new BizException("菜品库存不足: " + item.getDishName());
            }
        }

        // 4. 计算总金额
        BigDecimal total = items.stream()
            .map(i -> i.getPrice().multiply(BigDecimal.valueOf(i.getQuantity())))
            .reduce(BigDecimal.ZERO, BigDecimal::add);

        // 5. 优惠券抵扣（可选）
        BigDecimal discount = BigDecimal.ZERO;
        if (dto.getCouponId() != null) {
            discount = couponService.lock(dto.getCouponId(), total);
        }

        // 6. 写入订单 + 订单项
        Order order = new Order();
        order.setOrderNo(generateOrderNo());
        order.setTableId(dto.getTableId());
        order.setUserId(dto.getUserId());
        order.setTotalAmount(total);
        order.setPayAmount(total.subtract(discount));
        order.setStatus(OrderStatusEnum.PENDING_PAY.getCode());
        orderMapper.insert(order);

        // 7. 关联订单项
        for (CartItem item : items) {
            OrderItem oi = new OrderItem();
            oi.setOrderId(order.getId());
            oi.setDishId(item.getDishId());
            oi.setQuantity(item.getQuantity());
            oi.setPrice(item.getPrice());
            orderMapper.insertItem(oi);
        }

        // 8. 清空购物车
        cartMapper.deleteByUserId(dto.getUserId());

        return orderMapper.selectVOById(order.getId());
    }
}

@Service
@RequiredArgsConstructor
public class PaymentServiceImpl implements PaymentService {

    private final OrderMapper orderMapper;
    private final PaymentMapper paymentMapper;

    /**
     * 微信支付回调
     * 幂等性：同一个支付通知可能会来多次，必须能容忍
     */
    @Override
    @Transactional(rollbackFor = Exception.class)
    public void onPaySuccess(PayNotifyDTO notify) {
        // 1. 验签（先验证签名，再处理业务）
        if (!verifySignature(notify)) {
            throw new BizException("签名校验失败");
        }

        // 2. 查订单（关键：先查状态）
        Order order = orderMapper.selectByOrderNo(notify.getOrderNo());
        if (order == null) {
            throw new BizException("订单不存在");
        }

        // 3. 幂等性检查：如果订单已经是"已支付"，直接返回
        if (order.getStatus() == OrderStatusEnum.PAID.getCode()) {
            log.info("订单 {} 已处理过支付回调，跳过", order.getOrderNo());
            return;       // 关键：不要重复处理
        }

        // 4. 状态机推进：待支付 → 已支付
        if (order.getStatus() != OrderStatusEnum.PENDING_PAY.getCode()) {
            throw new BizException("订单状态异常: " + order.getStatus());
        }
        order.setStatus(OrderStatusEnum.PAID.getCode());
        order.setPayTime(LocalDateTime.now());
        orderMapper.updateById(order);

        // 5. 记录支付流水
        Payment payment = new Payment();
        payment.setOrderId(order.getId());
        payment.setAmount(order.getPayAmount());
        payment.setTransactionId(notify.getTransactionId());
        payment.setPayTime(LocalDateTime.now());
        paymentMapper.insert(payment);

        // 6. 后厨消息推送（异步）
        kitchenNotifyService.pushNewOrder(order);
    }
}

@Service
@RequiredArgsConstructor
public class MemberSettlementServiceImpl implements MemberSettlementService {

    private final MemberMapper memberMapper;
    private final PointsRecordMapper pointsMapper;
    private final GrowthRecordMapper growthMapper;
    private final MemberLevelMapper levelMapper;

    /**
     * 订单结算后调用：加积分 + 涨成长 + 检查升级
     */
    @Override
    @Transactional(rollbackFor = Exception.class)
    public void onOrderPaid(Order order) {
        Member member = memberMapper.selectByUserId(order.getUserId());
        if (member == null) return;        // 非会员不处理

        // 1. 积分：消费 1 元 = 1 积分
        int points = order.getPayAmount().intValue();
        member.setPoints(member.getPoints() + points);
        memberMapper.updateById(member);

        // 2. 记录积分流水（这是合规要求）
        PointsRecord pr = new PointsRecord();
        pr.setMemberId(member.getId());
        pr.setChange(points);
        pr.setType(PointsTypeEnum.ORDER_REWARD.getCode());
        pr.setOrderId(order.getId());
        pr.setRemark("订单奖励");
        pointsMapper.insert(pr);

        // 3. 成长值：消费 1 元 = 1 成长值
        int oldLevel = member.getLevel();
        member.setGrowth(member.getGrowth() + points);
        memberMapper.updateById(member);

        // 4. 检查升级（成长值跨阈值 → 自动升级）
        MemberLevel newLevel = levelMapper.selectByGrowth(member.getGrowth());
        if (newLevel != null && newLevel.getLevel() > oldLevel) {
            member.setLevel(newLevel.getLevel());
            memberMapper.updateById(member);

            // 5. 升级奖励：发券或发积分
            if (newLevel.getUpgradeCouponId() != null) {
                couponService.grant(member.getUserId(), newLevel.getUpgradeCouponId());
            }
        }

        // 6. 记录成长流水
        GrowthRecord gr = new GrowthRecord();
        gr.setMemberId(member.getId());
        gr.setChange(points);
        gr.setOrderId(order.getId());
        growthMapper.insert(gr);
    }
}

@Service
@RequiredArgsConstructor
public class CouponServiceImpl implements CouponService {

    private final UserCouponMapper userCouponMapper;

    /**
     * 阶段 1：领取（用户主动领）
     */
    @Override
    public void grant(Long userId, Long couponId) {
        // 检查库存、限领次数、有效期
        UserCoupon uc = new UserCoupon();
        uc.setUserId(userId);
        uc.setCouponId(couponId);
        uc.setStatus(CouponStatusEnum.UNUSED.getCode());
        uc.setExpireTime(calculateExpire(couponId));
        userCouponMapper.insert(uc);
    }

    /**
     * 阶段 2：锁定（下单时占用，避免多单共用一张券）
     */
    @Override
    public BigDecimal lock(Long userCouponId, BigDecimal orderAmount) {
        UserCoupon uc = userCouponMapper.selectById(userCouponId);
        if (uc.getStatus() != CouponStatusEnum.UNUSED.getCode()) {
            throw new BizException("券不可用");
        }
        // 状态：未使用 → 锁定
        uc.setStatus(CouponStatusEnum.LOCKED.getCode());
        userCouponMapper.updateById(uc);  // 用 updateById + 状态条件保证幂等
        return calculateDiscount(uc, orderAmount);
    }

    /**
     * 阶段 3：核销（支付成功后真正扣减）
     */
    @Override
    public void consume(Long userCouponId) {
        UserCoupon uc = userCouponMapper.selectById(userCouponId);
        uc.setStatus(CouponStatusEnum.USED.getCode());
        uc.setUseTime(LocalDateTime.now());
        userCouponMapper.updateById(uc);
    }

    /**
     * 回退：退款时释放券
     */
    @Override
    public void release(Long userCouponId) {
        UserCoupon uc = userCouponMapper.selectById(userCouponId);
        if (uc.getStatus() == CouponStatusEnum.LOCKED.getCode()) {
            uc.setStatus(CouponStatusEnum.UNUSED.getCode());
            userCouponMapper.updateById(uc);
        }
    }
}

/**
 * 订单退款时反向操作

@Override
@Transactional(rollbackFor = Exception.class)
public void onOrderRefunded(Order order) {
    Member member = memberMapper.selectByUserId(order.getUserId());
    if (member == null) return;

    int points = order.getPayAmount().intValue();

    // 1. 积分扣回（不能扣到负数）
    int newPoints = Math.max(0, member.getPoints() - points);
    member.setPoints(newPoints);
    memberMapper.updateById(member);

    // 2. 记录反向流水（必须有，便于审计）
    PointsRecord pr = new PointsRecord();
    pr.setMemberId(member.getId());
    pr.setChange(-points);
    pr.setType(PointsTypeEnum.ORDER_REFUND.getCode());
    pr.setOrderId(order.getId());
    pr.setRemark("订单退款扣回");
    pointsMapper.insert(pr);

    // 3. 成长值也回退（防止"刷成长"漏洞）
    member.setGrowth(Math.max(0, member.getGrowth() - points));
    memberMapper.updateById(member);

    // 4. 重新计算等级（可能降级）
    MemberLevel newLevel = levelMapper.selectByGrowth(member.getGrowth());
    if (newLevel != null) {
        member.setLevel(newLevel.getLevel());
        memberMapper.updateById(member);
    }
}

@Service
@RequiredArgsConstructor
public class ReliableMessageServiceImpl {

    private final RocketMQTemplate rocketMQTemplate;

    /**
     * 发送可靠消息（带本地消息表）
     */
    @Transactional
    public void sendReliableMessage(String topic, Object payload) {
        // 1. 写本地消息表（状态：待发送）
        LocalMessage msg = new LocalMessage();
        msg.setTopic(topic);
        msg.setPayload(JSON.toJSONString(payload));
        msg.setStatus(MessageStatusEnum.PENDING.getCode());
        localMessageMapper.insert(msg);

        // 2. 发到 MQ
        try {
            rocketMQTemplate.syncSend(topic, payload);
            msg.setStatus(MessageStatusEnum.SENT.getCode());
            localMessageMapper.updateById(msg);
        } catch (Exception e) {
            // 3. 失败等定时任务重试
            log.error("MQ 发送失败", e);
        }
    }
}
⚠️ 坑：小程序为什么不直接用 STOMP？ 微信小程序原生 wx.connectSocket 是 WebSocket 协议但不支持 STOMP 帧格式。常见替代方案：
  • 轮询：setInterval 每 5 秒请求一次（够用但费电）
  • SSE：服务端单向推送（仅适用"通知顾客"场景）
  • 自实现 STOMP 客户端：用 wx.sendSocketMessage 手写帧解析（成本高）
本项目选择关闭小程序 Socket，是务实的取舍。

@Component
public class RedisUtils {
    @Autowired
    private StringRedisTemplate redisTemplate;

    public void set(String key, Object value, long timeoutSeconds) {
        redisTemplate.opsForValue().set(key, JSON.toJSONString(value), timeoutSeconds, TimeUnit.SECONDS);
    }

    public <T> T get(String key, Class<T> clazz) {
        String json = redisTemplate.opsForValue().get(key);
        return json == null ? null : JSON.parseObject(json, clazz);
    }
}

// 1. 生产者：发送发券消息
@Autowired
private RocketMQTemplate rocketMQTemplate;

public void sendGrantTask(GrantTask task) {
    Message<GrantTask> msg = MessageBuilder.withPayload(task)
        .setHeader("KEYS", task.getUserId().toString())
        .build();
    rocketMQTemplate.syncSend("coupon-grant-topic:TASK_DISPATCH", msg);
    log.info("已发送发券消息 userId={}", task.getUserId());
}

// 2. 消费者：处理发券
@RocketMQMessageListener(
    topic = "coupon-grant-topic",
    selectorExpression = "TASK_DISPATCH",
    consumerGroup = "coupon-grant-dispatch-group"
)
public class GrantDispatchListener implements RocketMQListener<GrantTask> {

    @Override
    public void onMessage(GrantTask task) {
        log.info("收到发券任务 userId={}", task.getUserId());
        couponMapper.grantToUser(task.getUserId(), task.getCouponId());
    }
}

坑 1：消息堆积
现象：消费速度跟不上生产速度。

// 文件：com.scaffold.common.config.MinIOConfig.java
@Configuration
public class MinIOConfig {

    @Value("${storage.minio.endpoint}")
    private String endpoint;

    @Value("${storage.minio.access-key}")
    private String accessKey;

    @Value("${storage.minio.secret-key}")
    private String secretKey;

    @Bean
    public MinioClient minioClient() {
        return MinioClient.builder()
            .endpoint(endpoint)
            .credentials(accessKey, secretKey)
            .build();
    }
}

// 文件上传示例
@Service
@RequiredArgsConstructor
public class FileService {

    private final MinioClient minioClient;

    @Value("${storage.minio.bucket}")
    private String bucket;

    public String upload(MultipartFile file) throws Exception {
        // 1. 确保 bucket 存在（后端启动时会自动建）
        boolean found = minioClient.bucketExists(BucketExistsArgs.builder().bucket(bucket).build());
        if (!found) {
            minioClient.makeBucket(MakeBucketArgs.builder().bucket(bucket).build());
        }

        // 2. 上传
        String filename = UUID.randomUUID() + "-" + file.getOriginalFilename();
        minioClient.putObject(
            PutObjectArgs.builder()
                .bucket(bucket)
                .object(filename)
                .stream(file.getInputStream(), file.getSize(), -1)
                .contentType(file.getContentType())
                .build()
        );

        // 3. 返回可访问的 URL
        return endpoint + "/" + bucket + "/" + filename;
    }
}

A 概念辨析 docker compose down 和 docker compose down -v 有什么区别？
  • A. 没区别，都是停止容器

🏠 类比：你的 GitHub 账号是一栋大楼
  • User Pages = 大楼大厅（一个账号只能有一个，URL 固定 /<user>.github.io/）
  • Project Pages = 大楼里的独立房间（每个启用了 Pages 的仓库一个，URL /<repo>/）
