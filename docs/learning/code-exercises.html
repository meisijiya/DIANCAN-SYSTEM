后端：Service 关键逻辑

@Service
@RequiredArgsConstructor
public class DashboardServiceImpl implements DashboardService {
    private final OrderMapper orderMapper;

    @Override
    @Transactional
    public void advance(Long orderId) {
        Order order = orderMapper.selectById(orderId);
        if (order == null) throw new BusinessException("订单不存在");

        // 状态机：已支付 → 出餐中
        if (order.getStatus() != OrderStatusEnum.PAID.getCode()) {
            throw new BusinessException("只有已支付订单可以推进");
        }
        order.setStatus(OrderStatusEnum.SERVING.getCode());
        orderMapper.updateById(order);
    }

    @Override
    public Map<Integer, List<OrderVO>> groupByStatus() {
        List<OrderVO> all = orderMapper.selectList(null)
            .stream().map(this::toVO).collect(Collectors.toList());
        return all.stream().collect(Collectors.groupingBy(OrderVO::getStatus));
    }
}
